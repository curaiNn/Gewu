  /* ---------- 分子层：x、y 为容器内归一化坐标（0～1），速度单位 /min ---------- */
  function mol(S,type,x,y){const r=S.rng,th=r()*2*Math.PI,v=SPEED*Math.sqrt(S.T/T1)*(0.6+0.8*r());
    return{type,x:x??0.05+0.9*r(),y:y??0.05+0.9*r(),vx:v*Math.cos(th),vy:v*Math.sin(th),a:r()*2*Math.PI,w:(r()-0.5)*6,anim:null}}
  const count=(S,t)=>S.mols.reduce((n,m)=>n+(m.type===t),0);
  const busy=S=>S.mols.reduce((n,m)=>n+(m.anim?1:0),0);
  function inject(S,type,n,inlet){for(let i=0;i<n;i++){
    const m=inlet?mol(S,type,0.04,0.3+0.4*S.rng()):mol(S,type);if(inlet)m.vx=Math.abs(m.vx)+0.4;S.mols.push(m)}}
  // 2NO₂ → N₂O₄：随机挑一个 NO₂，与离它最近的 NO₂ 合成
  function forward(S,quiet){
    const A=S.mols.filter(m=>m.type==='A');if(A.length<2)return false;
    const a=A[Math.floor(S.rng()*A.length)];let b=null,d=Infinity;
    for(const m of A)if(m!==a){const e=(m.x-a.x)**2+(m.y-a.y)**2;if(e<d){d=e;b=m}}
    const n=mol(S,'B',(a.x+b.x)/2,(a.y+b.y)/2);
    n.vx=(a.vx+b.vx)/2;n.vy=(a.vy+b.vy)/2;n.a=Math.atan2(b.y-a.y,b.x-a.x);
    if(!quiet&&busy(S)<MAXANIM)n.anim={kind:'merge',t0:S.t,from:[[a.x,a.y,a.a],[b.x,b.y,b.a]]};
    S.mols=S.mols.filter(m=>m!==a&&m!==b);S.mols.push(n);return true}
  // N₂O₄ → 2NO₂：随机挑一个 N₂O₄ 拆开，两半反向弹开
  function reverse(S,quiet){
    const B=S.mols.filter(m=>m.type==='B');if(!B.length)return false;
    const b=B[Math.floor(S.rng()*B.length)],c=Math.cos(b.a),s=Math.sin(b.a),v=SPEED*Math.sqrt(S.T/T1);
    const p=mol(S,'A',b.x-c*0.015,b.y-s*0.015),q=mol(S,'A',b.x+c*0.015,b.y+s*0.015);
    p.vx=b.vx-c*v;p.vy=b.vy-s*v;q.vx=b.vx+c*v;q.vy=b.vy+s*v;
    if(!quiet&&busy(S)<MAXANIM)p.anim={kind:'split',t0:S.t,x:b.x,y:b.y};
    S.mols=S.mols.filter(m=>m!==b);S.mols.push(p,q);return true}
  function move(S,h){for(const m of S.mols){
    m.x+=m.vx*h;m.y+=m.vy*h;m.a+=m.w*h;
    if(m.x<0.03){m.x=0.03;m.vx=Math.abs(m.vx)}else if(m.x>0.97){m.x=0.97;m.vx=-Math.abs(m.vx)}
    if(m.y<0.03){m.y=0.03;m.vy=Math.abs(m.vy)}else if(m.y>0.97){m.y=0.97;m.vy=-Math.abs(m.vy)}
    if(m.anim&&S.t-m.anim.t0>=ANIM)m.anim=null}}
  // 让可见分子数与浓度对齐（初始化与扰动后）：N₂O₄ 取整，NO₂ 由氮守恒定；增减不播动画
  function align(S){
    const s=S.V*S.s,wB=Math.round(S.cB*s);
    while(count(S,'B')>wB)reverse(S,true);
    while(count(S,'B')<wB){if(count(S,'A')>=2)forward(S,true);else inject(S,'B',1,false)}
    let d=Math.max(0,Math.round((S.cA+2*S.cB)*s)-2*wB)-count(S,'A');
    if(d>0)inject(S,'A',d,false);
    for(;d<0;d++)S.mols.splice(S.mols.findIndex(m=>m.type==='A'),1);
    S.F=0.5-(wB-S.cB*s);S.Rr=0.5}    // 此后 N₂O₄ 个数差 = Rr − F ∈ (−1, 1)
